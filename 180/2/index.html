def convolute_4(input, kernel, padding="same"):

    # check if kernel is 2D and odd in both dims (so there is a center pixel)
    if kernel.ndim != 2:
        raise ValueError(f"kernel must be 2D, got shape {kernel.shape}")
    if any(s % 2 == 0 for s in kernel.shape):
        raise ValueError(f"kernel dims must be odd, got shape {kernel.shape}")

    kh, kw = kernel.shape

    #assigning padding size, separately per axis
    if padding == "same":
        ph, pw = (kh - 1)//2, (kw - 1)//2
    elif padding == "full":
        ph, pw = kh - 1, kw - 1
    else:
        raise ValueError(f"padding must be 'same' or 'full', got {padding!r}")

    k_flip = np.flip(kernel, (0,1))
    in_padded = np.pad(input, ((ph, ph), (pw, pw)), constant_values=0)
    H, W = input.shape
    out_h = H + 2 * ph - kh + 1
    out_w = W + 2 * pw - kw + 1
    out = np.zeros((out_h, out_w), dtype=float)

    # offsets run -half .. +half around the window center, one half per axis
    half_h, half_w = (kh - 1)//2, (kw - 1)//2

    for m in range(half_h, in_padded.shape[0] - half_h):
        for n in range(half_w, in_padded.shape[1] - half_w):
            for i in range(-half_h, half_h + 1):
                for j in range(-half_w, half_w + 1):
                    out[m - half_h, n - half_w] += in_padded[m + i,n + j] * k_flip[i + half_h, j + half_w]

    return out
convolute_2 — two loops 
def convolute_2(input, kernel, padding="same"):

    # check if kernel is 2D and odd in both dims (so there is a center pixel)
    if kernel.ndim != 2:
        raise ValueError(f"kernel must be 2D, got shape {kernel.shape}")
    if any(s % 2 == 0 for s in kernel.shape):
        raise ValueError(f"kernel dims must be odd, got shape {kernel.shape}")

    kh, kw = kernel.shape

    #assigning padding size, separately per axis
    if padding == "same":
        ph, pw = (kh - 1)//2, (kw - 1)//2
    elif padding == "full":
        ph, pw = kh - 1, kw - 1
    else:
        raise ValueError(f"padding must be 'same' or 'full', got {padding!r}")

    k_flip = np.flip(kernel, (0,1))
    in_padded = np.pad(input, ((ph, ph), (pw, pw)), constant_values=0)
    H, W = input.shape
    out_h = H + 2 * ph - kh + 1
    out_w = W + 2 * pw - kw + 1
    out = np.zeros((out_h, out_w), dtype=float)

    # offsets run -half .. +half around the window center, one half per axis
    half_h, half_w = (kh - 1)//2, (kw - 1)//2

    for m in range(in_padded.shape[0] - kh + 1):
        for n in range(in_padded.shape[1] - kw + 1):
            patch = in_padded[m: m + kh, n: n + kw]
            out[m,n] = np.sum(patch * k_flip)

    return out

Runtime

The 4 for loops is in n^4 runtime where n is a dimension of the image because of the 4 for loops. The 2 for loops is a vectorized version but it is still slow on big images. The builtin scipy convolve2d is much faster taking only seconds to convolve big images.

Boundaries

With no padding, convolution shrinks an image because the kernel would align its ediges to the image edges. I implemented two sizes of padding one that keeps the image the same size and another that keeps all possible kernel and image overlap.
